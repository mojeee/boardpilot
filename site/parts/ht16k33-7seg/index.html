Starter code (Arduino)

#include <Wire.h>

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);  // SDA = GPIO 21, SCL = GPIO 22
  Wire.beginTransmission(0x70);
  bool found = Wire.endTransmission() == 0;
  Serial.println(found ? "HT16K33 4-digit 7-segment backpack (I2C) found at 0x70" : "Not found: check power, GND and SDA/SCL");
}

void loop() {}
