Starter code (Arduino)

#include <Wire.h>

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);  // SDA = GPIO 21, SCL = GPIO 22
  Wire.beginTransmission(0x70);
  bool found = Wire.endTransmission() == 0;
  Serial.println(found ? "TCA9548A I2C multiplexer (8 channels) found at 0x70" : "Not found: check power, GND and SDA/SCL");
}

void loop() {}
